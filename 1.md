import asyncio
import aiohttp  # 这才是真正的 HTTP 客户端库
#
# 这份代码原本是现在网站上的图片的，想下载本地启动服务，因路径太短抛出异常，下载有修正版本
#
async def download_file(session, url):
    print(f"开始下载{url}")
    # 发送请求
    response = await session.get(url)
    # 等待数据
    content = await response.read()
    filename = url.split('/')[-1]
    print("下载完毕")
    # 保存图片到本地
    with open(filename, "wb") as f:
        f.write(content)
    # 释放资源
    await response.release()


async def main():
    url = ["http://127.0.0.1:8003/static/cat.jpeg",
           "http://127.0.0.1:8003/static/lion.jpeg",
           "http://127.0.0.1:8003/static/man.png"]
    # 创建会话对象
    session = aiohttp.ClientSession()
    # 创建多个协程
    coroutine_list = [download_file(session, url[i]) for i in range(len(url))]
    # 交给事件循环
    await asyncio.gather(*coroutine_list)
    # 关闭会话
    await session.close()


if __name__ == '__main__':
    asyncio.run(main())


# 修正，加完善版本
import asyncio
import aiohttp

async def download_file(session, url):
    print(f"开始下载 {url}")
    try:
        async with session.get(url) as response:
            content = await response.read()
            # 从 URL 中提取文件名
            filename = url.split('/')[-1]
            with open(filename, "wb") as f:
                f.write(content)
            print(f"下载完毕: {filename}")
    except Exception as e:
        print(f"下载 {url} 失败: {e}")

async def main():
    urls = [
        "http://127.0.0.1:8003/static/cat.jpeg",
        "http://127.0.0.1:8003/static/lion.jpeg",
        "http://127.0.0.1:8003/static/man.png"
    ]
    async with aiohttp.ClientSession() as session:
        tasks = [download_file(session, url) for url in urls]
        await asyncio.gather(*tasks)

if __name__ == '__main__':
    asyncio.run(main())
